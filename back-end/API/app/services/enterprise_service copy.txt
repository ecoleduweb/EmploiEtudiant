    def updateEnterprise(self, data):
        enterprise_obj = Enterprise(**data)
        
        current_employers = self.getAllEmployersFromEntreprise(enterprise_obj.id) or []
        current_user_ids = [e.userId for e in current_employers]
        
        new_user_ids = [u['id'] if isinstance(u, dict) else u.id for u in (enterprise_obj.users or [])]

        user_ids_to_remove = [
            uid for uid in current_user_ids
            if uid not in new_user_ids
        ]

        return enterprise_repo.updateEnterprise(enterprise_obj, user_ids_to_remove)
