    def updateEnterprise(self, entreprise: Enterprise, user_ids_to_remove):
        try:
            enterprise = Enterprise.query.get(entreprise.id)
            if not enterprise:
                return None

            enterprise.name = entreprise.name
            enterprise.email = entreprise.email
            enterprise.phone = entreprise.phone
            enterprise.address = entreprise.address
            enterprise.cityId = entreprise.cityId

            if user_ids_to_remove:
                employers_to_update = Employers.query.filter(
                    Employers.userId.in_(user_ids_to_remove),
                    Employers.enterpriseId == enterprise.id
                ).all()

                for emp in employers_to_update:
                    emp.enterpriseId = None

            for user in entreprise.users or []:
                employer = Employers.query.filter_by(userId=user.id).first()

                if employer:
                    employer.enterpriseId = enterprise.id
                else:
                    logger.warning("user not found")
            db.session.commit()
            return enterprise

        except Exception as e:
            db.session.rollback()
            raise e
    def updateUserEnterprise(self, enterprise_id, users_list):
        try:
            Employers.query.filter_by(enterpriseId=enterprise_id).delete()
            
            for u in users_list:
                ##u_id = u.get('id') if isinstance(u, dict) else u
                u_id = u.id if hasattr(u, 'id') else u 
                if u_id:
                    new_employer = Employers(
                        userId=u_id, 
                        enterpriseId=enterprise_id, 
                        verified=False
                    )
                    db.session.add(new_employer)
            db.session.flush()
        except Exception as e:
            raise e
    def deleteEnterprise(self, id):
        enterprise = Enterprise.query.filter_by(id=id).first()
        if enterprise.isTemporary == True:
            db.session.delete(enterprise)
            db.session.commit()
        else:
            logger.error('enterprise is not temporary')
            return False
        logger.warning('enterprise deleted')
        return True
